# Izanagi Snapshot Format and Structure

## 1. Purpose

Izanagi creates a deterministic, local-first project context snapshot for use with AI assistants.

A snapshot captures the project's relevant files, structure, configuration, and metadata at a particular point in time. It is intended to provide reusable, inspectable project context without requiring the assistant to rediscover the repository in every conversation.

Izanagi does not execute project code or generate AI summaries. Eligible file contents are preserved verbatim.

A snapshot describes the project as captured; it does not guarantee that the project remains unchanged afterward.

## 2. Snapshot Outputs

A snapshot consists of:

- A Markdown context document containing project metadata, inventory, directory structure, and eligible file contents.
- A JSON manifest containing structured inventory and snapshot metadata.

The snapshot output directory is managed by Izanagi and must not be treated as ordinary project input during the same scan.

The snapshot header records the metadata available to the implementation, including:

- Project name
- Snapshot ID
- Creation timestamp, including timezone information
- Resolved project root
- Git root, when available
- Snapshot format version
- Izanagi package version, when available

The JSON manifest records the snapshot format version and structured project inventory.

Metadata fields that cannot be determined should be represented as unavailable rather than fabricated.

def _update_current(source: Path, output_root: Path) -> Path:
    """Atomically publish the latest successful Izanagi output as current.md."""
    output_root.mkdir(parents=True, exist_ok=True)
    current_path = output_root / "current.md"

    file_descriptor, temporary_name = tempfile.mkstemp(
        prefix=".current-",
        suffix=".tmp",
        dir=output_root,
    )
    temporary_path = Path(temporary_name)

    try:
        with os.fdopen(file_descriptor, "wb") as temporary_file:
            temporary_file.write(source.read_bytes())
            temporary_file.flush()
            os.fsync(temporary_file.fileno())

        os.replace(temporary_path, current_path)
    except BaseException:
        temporary_path.unlink(missing_ok=True)
        raise

    return current_path


def _get_char_limit(config: dict) -> int:
    """Read the segmentation limit from the [context] configuration."""
    value = config.get("context", {}).get("char_limit", 5000)

    if isinstance(value, bool) or not isinstance(value, int) or value < 1:
        raise ValueError(
            "[context].char_limit must be a positive integer"

    return value

from __future__ import annotations

import base64
import hashlib
import json
import uuid
from datetime import UTC, datetime
from importlib.metadata import PackageNotFoundError, version
from pathlib import Path
from typing import Any

from izanagi.manifest import manifest_project

FORMAT_VERSION = 1


def _izanagi_version() -> str | None:
    """Return the installed package version, if available."""
    try:
        return version("izanagi")
    except PackageNotFoundError:
        return None


def _format_bytes(data: bytes) -> str:
    """Represent file bytes as UTF-8 when possible, otherwise as Base64."""
    try:
        return data.decode("utf-8")
    except UnicodeDecodeError:
        return "Base64-encoded binary file\n\n" + base64.b64encode(data).decode(
            "ascii"
        )


def _is_utf8(data: bytes) -> bool:
    """Return whether bytes can be decoded as UTF-8."""
    try:
        data.decode("utf-8")
    except UnicodeDecodeError:
        return False
    return True


def _fence_for(content: str) -> str:
    """Choose a Markdown fence longer than any backtick run in the content."""
    longest = 0
    current = 0

    for character in content:
        if character == "`":
            current += 1
            longest = max(longest, current)
        else:
            current = 0

    return "`" * max(3, longest + 1)


def _directory_structure(records: list[dict[str, Any]]) -> str:
    """Render a deterministic tree, pruning excluded directories without included descendants."""
    nodes: dict[str, dict[str, Any]] = {}

    for record in records:
        path = record["path"].rstrip("/")
        if not path:
            continue

        parts = path.split("/")
        for index in range(len(parts)):
            relative = "/".join(parts[: index + 1])
            is_explicit_record = index == len(parts) - 1
            existing = nodes.get(relative)

            if existing is None:
                nodes[relative] = {
                    "path": relative,
                    "name": parts[index],
                    "kind": (
                        record["kind"] if is_explicit_record else "directory"
                    ),
                    "included": (
                        bool(record["included"]) if is_explicit_record else False
                    ),
                    "explicit": is_explicit_record,
                }
            elif is_explicit_record:
                existing.update(
                    {
                        "kind": record["kind"],
                        "included": bool(record["included"]),
                        "explicit": True,
                    }
                )

    children: dict[str, list[str]] = {}
    for path in nodes:
        parent = path.rpartition("/")[0]
        children.setdefault(parent, []).append(path)

    for child_paths in children.values():
        child_paths.sort(key=lambda item: (nodes[item]["name"].casefold(), item))

    # A directory is visible if it is included, is not explicitly excluded,
    # or contains an included descendant that is a specific inclusion override.
    visible: set[str] = set()
    for path, node in nodes.items():
        if node["kind"] != "directory":
            if node["included"]:
                visible.add(path)
            continue

        descendants = [
            other
            for other in nodes
            if other.startswith(path + "/") and nodes[other]["included"]
        ]
        explicitly_excluded = node["explicit"] and not node["included"]
        if not explicitly_excluded or descendants:
            visible.add(path)

    lines: list[str] = []

    def render(parent: str, prefix: str = "") -> None:
        child_paths = [
            path
            for path in children.get(parent, [])
            if path in visible
        ]

        for index, path in enumerate(child_paths):
            node = nodes[path]
            is_last = index == len(child_paths) - 1
            connector = "└── " if is_last else "├── "
            name = node["name"]

            if node["kind"] == "directory":
                name += "/"

            lines.append(f"{prefix}{connector}{name}")
            if node["kind"] == "directory":
                render(path, prefix + ("    " if is_last else "│   "))

    render("")
    return "\n".join(lines) if lines else "(no included paths)"


def build_snapshot(project: Path, *, config: dict[str, Any]) -> dict[str, Any]:
    """Build base Markdown and a JSON-serializable manifest.

    Project code is never executed. Included file bytes are captured by
    manifest_project() and reused here for content, size, and SHA-256.
    Excluded files are not read or hashed.
    """
    inventory = manifest_project(project, config=config, capture_content=True)
    root = Path(inventory["project_root"])
    timestamp = datetime.now(UTC).isoformat()
    snapshot_id = str(uuid.uuid4())
    izanagi_version = _izanagi_version()

    records: list[dict[str, Any]] = []
    document_sections: list[tuple[str, str]] = []
    source_sections: list[tuple[str, str]] = []

    for original in inventory["files"]:
        record = dict(original)
        data = record.pop("_content", None)

        if (
            not record["included"]
            or record["kind"] in {"directory", "symlink", "unreadable"}
        ):
            record.pop("sha256", None)
            record.pop("encoding", None)
            if not record["included"]:
                record["size_bytes"] = None
            records.append(record)
            continue

        if data is None:
            record["included"] = False
            record["reason"] = "included file content was not captured"
            record["size_bytes"] = None
            record.pop("sha256", None)
            records.append(record)
            continue

        digest = hashlib.sha256(data).hexdigest()
        size = len(data)
        encoding = "utf-8" if _is_utf8(data) else "base64"

        record["size_bytes"] = size
        record["sha256"] = digest
        record["encoding"] = encoding
        records.append(record)

        content = _format_bytes(data)
        fence = _fence_for(content)
        section = (
            f"### `{record['path']}`\n\n"
            f"- Type: {record['kind']}\n"
            f"- Size: {size} bytes\n"
            f"- SHA-256: `{digest}`\n"
            f"- Encoding: {encoding}\n\n"
            f"{fence}\n{content}\n{fence}"
        )

        if record["kind"] == "documentation":
            document_sections.append((record["path"], section))
        else:
            source_sections.append((record["path"], section))

    records.sort(key=lambda item: item["path"])
    document_sections.sort(key=lambda item: item[0])
    source_sections.sort(key=lambda item: item[0])

    project_name = config.get("project", {}).get("name", "").strip()
    project_name = project_name or root.name

    preamble = config.get("preamble", {}).get("base_preamble", "").strip()
    header = [
        "# Izanagi Project — Base",
        "",
        f"- Project: {project_name}",
        f"- Base ID: `{snapshot_id}`",

        path
        in (

def classify_file(path: Path) -> str:
    """Classify a file for reporting; classification does not control inclusion."""
    name = path.name
    suffix = path.suffix.lower()

    if name in {
        "pyproject.toml",
        "requirements.txt",
        "uv.lock",
        "Makefile",
        "Dockerfile",
        "Cargo.toml",
        "CMakeLists.txt",
    }:
        return "config"
    if suffix in {
        ".py", ".c", ".cc", ".cpp", ".h", ".hpp", ".rs",
        ".sv", ".svh", ".v", ".vh", ".sh",
    }:
        return "source"
    if suffix == ".ipynb":
        return "notebook"
    if suffix in {".md", ".rst", ".txt"}:
        return "documentation"
    if suffix in {".toml", ".json", ".yaml", ".yml", ".ini", ".cfg"}:
        return "config"
    if suffix in {
        ".png", ".jpg", ".jpeg", ".gif", ".webp", ".pdf",
        ".zip", ".gz", ".pt", ".pth", ".bin",
    }:
        return "binary_or_artifact"
    return "other"


def _file_match(pattern: str, relative: str) -> bool:
    """Slash-free patterns match basenames at any depth."""
    pattern = pattern.replace("\\", "/")
    relative = relative.replace("\\", "/")

    if pattern.startswith("/"):
        return fnmatch.fnmatchcase(relative, pattern[1:])
    if "/" not in pattern:
        return fnmatch.fnmatchcase(Path(relative).name, pattern)
    return fnmatch.fnmatchcase(relative, pattern)


def _directory_match(pattern: str, relative: str) -> bool:
    """Match a directory rule against a directory or its descendants."""
    pattern = pattern.replace("\\", "/")
    anchored = pattern.startswith("/")
    pattern = pattern.removeprefix("/").rstrip("/")
    relative = relative.replace("\\", "/").strip("/")

    if not pattern or not relative:
        return False

    parts = relative.split("/")
    prefixes = ["/".join(parts[:i]) for i in range(1, len(parts) + 1)]

    if anchored:
        return any(fnmatch.fnmatchcase(prefix, pattern) for prefix in prefixes)

    if "/" not in pattern:
        return any(fnmatch.fnmatchcase(part, pattern) for part in parts)

    return any(
        fnmatch.fnmatchcase(prefix, pattern)
        for prefix in prefixes
        if "/" in prefix
    )


def _rule_matches(rule: str, relative: str) -> bool:
    """Test a path against a directory rule or file/pattern rule."""
    if rule.endswith("/"):
        return _directory_match(rule, relative)
    return _file_match(rule, relative)


def _rule_specificity(rule: str) -> tuple[int, int]:
    """Rank exact file paths above directory rules and broad patterns.

    Tier 3: literal file paths.
    Tier 2: directory rules.
    Tier 1: wildcard filename or path patterns.

    Within each tier, deeper paths are more specific.
    """
    normalized = rule.replace("\\", "/").removeprefix("/")
    has_wildcards = any(char in normalized for char in "*?[")

    if not normalized.endswith("/") and not has_wildcards:
        return (3, normalized.count("/"))

    if normalized.endswith("/"):
        return (2, normalized.rstrip("/").count("/") + 1)

    return (1, normalized.count("/"))


def inclusion_decision(
    path: Path,
    *,
    inclusion_rules: list[str],
    exclusion_rules: list[str],
) -> tuple[bool, str]:
    """Apply the most-specific matching rule; exclusions win ties."""
    relative = path.as_posix().replace("\\", "/").strip("/")

    matches: list[tuple[tuple[int, int], bool, str]] = []

    for included, rules in (
        (True, inclusion_rules),
        (False, exclusion_rules),
    ):
        for rule in rules:
            if _rule_matches(rule, relative):
                matches.append((_rule_specificity(rule), included, rule))

    if not matches:
        return False, "no inclusion rule matched"

    highest_specificity = max(match[0] for match in matches)
    winners = [match for match in matches if match[0] == highest_specificity]

    # Exclusion wins when rules have equal specificity.
    winner = next(
        (match for match in winners if not match[1]),
        winners[0],
    )

    included, rule = winner[1], winner[2]
    decision = "included" if included else "excluded"
    return included, f"{decision} by rule: {rule}"


def _should_traverse_excluded_directory(
    relative: str,
    *,
    inclusion_rules: list[str],
) -> bool:
    """Determine whether an excluded directory may contain an override.

    Only a more-specific inclusion can justify traversing an excluded
    directory. Broad filename patterns alone do not override directory
    exclusions.
    """
    directory = relative.replace("\\", "/").strip("/")
    prefix = directory + "/"
    directory_specificity = (2, directory.count("/") + 1)

    for rule in inclusion_rules:
        normalized = rule.replace("\\", "/").removeprefix("/")
        rule_specificity = _rule_specificity(rule)

        # An explicit file path under this directory can override it.
        if (
            not rule.endswith("/")
            and not any(char in normalized for char in "*?[")
            and normalized.startswith(prefix)
            and rule_specificity > directory_specificity
        ):
            return True

        # A narrower included directory can override this directory.
        if (
            rule.endswith("/")
            and normalized.startswith(prefix)
            and rule_specificity > directory_specificity
        ):
            return True

    return False


def discover_git_root(project_root: Path) -> Path | None:
    """Find the nearest ancestor containing a .git entry without invoking Git."""
    for candidate in (project_root, *project_root.parents):
        if (candidate / ".git").exists():
            return candidate
    return None


def manifest_project(
    project: Path,
    *,
    config: dict[str, Any],
    capture_content: bool = False,
) -> dict[str, Any]:
    """Build a project manifest using configured inclusion and exclusion rules.

    Excluded files are not read. Included files are read once to determine
    their size and hash. When capture_content=True, their bytes are attached
    under the private '_content' key for snapshot generation to reuse.

    Generated Izanagi output files are skipped. The local config file remains
    eligible for inclusion.

    Callers must remove '_content' before serializing the manifest to JSON.
    """
    root = project.expanduser().resolve(strict=True)
    if not root.is_dir():
        raise NotADirectoryError(f"Project path is not a directory: {root}")

    scan = config["scan"]
    inclusion_rules = scan["inclusions"]["rules"]
    exclusion_rules = scan["exclusions"]["rules"]
    include_hidden = scan.get("include_hidden_files", True)
    records: list[dict[str, Any]] = []

    def walk_error(error: OSError) -> None:
        records.append(
            {
                "path": str(error.filename or "<unknown>"),
                "kind": "unreadable",
                "included": False,
                "reason": f"directory traversal error: {error}",
                "size_bytes": None,
            }
        )

    for current, dirs, filenames in os.walk(
        root,
        topdown=True,
        followlinks=False,
        onerror=walk_error,
    ):
        current_path = Path(current)
        dirs[:] = sorted(dirs)
        kept_dirs: list[str] = []

        for name in dirs:
            path = current_path / name
            relative = path.relative_to(root).as_posix()

            if _is_izanagi_output(relative):
                continue

            if path.is_symlink():
                records.append(
                    {
                        "path": relative,
                        "kind": "symlink",
                        "included": False,
                        "reason": "directory symlink not followed",
                        "size_bytes": None,
                    }
                )
                continue

            if (
                not include_hidden
                and any(part.startswith(".") for part in Path(relative).parts)
                and not any(
                    rule.endswith("/") and _directory_match(rule, relative)
                    for rule in inclusion_rules
                )
                and not any(
                    not rule.endswith("/") and _file_match(rule, relative)
                    for rule in inclusion_rules
                )
            ):
                continue

            excluded_rule = next(
                (
                    rule
                    for rule in exclusion_rules
                    if rule.endswith("/") and _directory_match(rule, relative)
                ),
                None,
            )

            if excluded_rule:
                records.append(
                    {
                        "path": relative + "/",
                        "kind": "directory",
                        "included": False,
                        "reason": f"excluded by rule: {excluded_rule}",
                        "size_bytes": None,
                    }
                )

                if _should_traverse_excluded_directory(
                    relative,
                    inclusion_rules=inclusion_rules,
                ):
                    kept_dirs.append(name)

                continue

            kept_dirs.append(name)

        dirs[:] = kept_dirs

        for filename in sorted(filenames):
            path = current_path / filename
            relative_path = path.relative_to(root)
            relative = relative_path.as_posix()

            # Generated output must never become a project delta.
            if _is_izanagi_output(relative):
                continue

            if path.is_symlink():
                records.append(
                    {
                        "path": relative,
                        "kind": "symlink",
                        "included": False,
                        "reason": "symlink not followed",
                        "size_bytes": None,
                    }
                )
                continue

            if (
                not include_hidden
                and any(part.startswith(".") for part in relative_path.parts)
                and not any(
                    rule.endswith("/") and _directory_match(rule, relative)
                    for rule in inclusion_rules
                )
                and not any(
                    not rule.endswith("/") and _file_match(rule, relative)
                    for rule in inclusion_rules
                )
            ):
                records.append(
                    {
                        "path": relative,
                        "kind": classify_file(path),
                        "included": False,
                        "reason": "hidden file excluded by scan setting",
                        "size_bytes": None,
                    }
                )
                continue

            included, reason = inclusion_decision(
                relative_path,
                inclusion_rules=inclusion_rules,
                exclusion_rules=exclusion_rules,
            )

            record: dict[str, Any] = {
                "path": relative,
                "kind": classify_file(path),
                "included": included,
                "reason": reason,
                "size_bytes": None,
            }

            # Avoid reading excluded file contents.
            if not included:
                records.append(record)
                continue

            try:
                content = path.read_bytes()
            except OSError as error:
                record["kind"] = "unreadable"
                record["included"] = False
                record["reason"] = f"file read error: {error}"
                records.append(record)
                continue

            record["size_bytes"] = len(content)
            record["sha256"] = hashlib.sha256(content).hexdigest()

            if capture_content:
                record["_content"] = content

            records.append(record)

    records.sort(key=lambda record: record["path"])

    file_records = [
        record for record in records if record["kind"] != "directory"
    ]
    included_records = [record for record in file_records if record["included"]]

    categories: dict[str, int] = {}
    for record in included_records:
        category = record["kind"]
        categories[category] = categories.get(category, 0) + 1

    known_sizes = [
        record["size_bytes"]
        for record in included_records
        if record["size_bytes"] is not None
    ]

    return {
        "project_root": str(root),
        "git_root": (
            str(git_root) if (git_root := discover_git_root(root)) else None
        ),
        "files": records,
        "summary": {
            "file_count": len(file_records),
            "included_count": len(included_records),
            "excluded_count": len(file_records) - len(included_records),
            "included_bytes": sum(known_sizes),
            "categories": categories,
        },
    }

"""Lossless, in-memory context segmentation and interactive clipboard copying."""

from __future__ import annotations

import hashlib
import os
import sys
from pathlib import Path
from typing import Callable

CONCLUSION = """<CONTEXT_CONCLUSION>
All segments in this context sequence have now been provided.
The complete source context is available by combining the payloads in
segment order. The headers, footers, and this conclusion are metadata,
not part of the original source. You may now analyze the complete context.
<END_CONTEXT_CONCLUSION>"""


def split_context(content: str, char_limit: int) -> list[str]:
    """Split text into ordered, lossless payloads no longer than char_limit.

    Splits prefer newline boundaries. If no newline is available within the
    limit, the text is split at the character limit.
    """
    if char_limit < 1:
        raise ValueError("char_limit must be at least 1")

    if not content:
        return [""]

    segments: list[str] = []
    start = 0

    while start < len(content):
        end = min(start + char_limit, len(content))

        if end < len(content):
            newline = content.rfind("\n", start, end)
            if newline >= start:
                # Include the newline in this segment so reconstruction is exact.
                end = newline + 1

        # A newline immediately at the start is still progress, and a long
        # line without newlines falls back to the hard character limit.
        if end <= start:
            end = min(start + char_limit, len(content))

        segments.append(content[start:end])
        start = end

    if "".join(segments) != content:
        raise AssertionError("Segmentation failed to preserve source content")

    if any(len(segment) > char_limit for segment in segments):
        raise AssertionError("A segment exceeds the configured character limit")

    return segments


def format_segments(
    content: str,
    char_limit: int,
    project_name: str,
) -> list[str]:
    """Create formatted clipboard segments without writing files.

    The source payloads remain unchanged and reconstruct the original content
    exactly. Headers, footers, and the final conclusion are additional text.
    """
    payloads = split_context(content, char_limit)
    context_id = hashlib.sha256(content.encode("utf-8")).hexdigest()[:16]
    total_segments = len(payloads)
    source_characters = len(content)

    formatted: list[str] = []

    for index, payload in enumerate(payloads, start=1):
        if index < total_segments:
            sequence_status = "More segments follow."
            remaining = total_segments - index
            completion = f"Segments remaining after this one: {remaining}."
        else:
            sequence_status = "This is the final segment."
            completion = "No further segments follow this one."

        header = (
            "<IZANAGI_CONTEXT>\n"
            f"Project: {project_name}\n"
            f"Context-ID: {context_id}\n"
            f"Segment: {index}/{total_segments}\n"
            f"Total source characters: {source_characters}\n"
            f"Segment payload characters: {len(payload)}\n"
            "Sequence: ordered; combine payloads in segment order\n"
            "Instructions: retain this segment and wait for all segments "
            "before analyzing the complete context.\n"
            "<END_HEADER>\n"
        )

        footer = (
            f"\n<SEGMENT_STATUS>\n"
            f"End of segment {index}/{total_segments}.\n"
            f"{sequence_status}\n"
            f"{completion}\n"
            "Treat this as partial context until the full sequence "
            "has been received.\n"
            "<END_SEGMENT>"
        )

        segment_text = header + payload + footer

        if index == total_segments:
            segment_text += "\n\n" + CONCLUSION

        formatted.append(segment_text)

    # Verify the payloads are still lossless after formatting.
    recovered_payloads: list[str] = []
    for segment_text, payload in zip(formatted, payloads, strict=True):
        header_end = segment_text.index("<END_HEADER>\n") + len("<END_HEADER>\n")
        footer_start = segment_text.index("\n<SEGMENT_STATUS>\n", header_end)
        recovered_payloads.append(segment_text[header_end:footer_start])
        if recovered_payloads[-1] != payload:
            raise AssertionError("Formatted segment changed its source payload")

    if "".join(recovered_payloads) != content:
        raise AssertionError("Formatted segments do not reconstruct the source")

    return formatted


def total_clipboard_characters(segments: list[str]) -> int:
    """Return the total characters across all formatted clipboard segments."""
    return sum(len(segment) for segment in segments)


def _read_single_key() -> str:
    """Read one key without requiring Enter, on Windows or Unix-like systems."""
    if os.name == "nt":
        import msvcrt

        return msvcrt.getwch()

    import termios
    import tty

    if not sys.stdin.isatty():
        raise RuntimeError(
            "Interactive segment copying requires a terminal. "
            "Run this command in a terminal."
        )

    file_descriptor = sys.stdin.fileno()
    previous_settings = termios.tcgetattr(file_descriptor)

    try:
        tty.setraw(file_descriptor)
        return sys.stdin.read(1)
    finally:
        termios.tcsetattr(file_descriptor, termios.TCSADRAIN, previous_settings)


def copy_segments_interactively(
    content: str,
    char_limit: int,
    project_name: str,
    copy_to_clipboard: Callable[[str], None],
) -> None:
    """Prepare all segments, report exact counts, then copy on Space presses."""
    segments = format_segments(content, char_limit, project_name)
    source_characters = len(content)
    clipboard_characters = total_clipboard_characters(segments)

    print(f"Source characters: {source_characters:,}")
    print(f"Payload character limit: {char_limit:,}")
    print(f"Segments: {len(segments):,}")
    print(f"Total clipboard characters across all segments: "
          f"{clipboard_characters:,}")
    print("Headers, footers, and the final conclusion are included in the "
          "clipboard character total.")
    print()

    for index, segment in enumerate(segments, start=1):
        prompt = (
            f"Press SPACE to copy segment {index}/{len(segments)}, "
            "or Q to quit: "
        )
        print(prompt, end="", flush=True)

        while True:
            key = _read_single_key()
            if key in (" ", "q", "Q", "\x03", "\x1b"):
                break

        if key in ("q", "Q", "\x03", "\x1b"):
            print("\nStopped. Remaining segments were not copied.")
            return

        try:
            copy_to_clipboard(segment)
        except Exception:
            print("\nClipboard copy failed.", file=sys.stderr)
            raise

        print(
            f"\rCopied segment {index}/{len(segments)} "
            f"({len(segment):,} clipboard characters)."
        )

    print("All segments copied.")

from izanagi.base import build_snapshot, write_snapshot


class SnapshotTests(unittest.TestCase):
    def setUp(self):
        self.temp_dir = tempfile.TemporaryDirectory()
        self.addCleanup(self.temp_dir.cleanup)
        self.root = Path(self.temp_dir.name) / "project"
        self.root.mkdir()
        self.config = {
            "project": {"name": "test-project"},
            "scan": {
                "include_hidden_files": True,
                "inclusions": {"rules": ["*.py", "*.md"]},
                "exclusions": {"rules": ["private.py"]},
            },
            "preamble": {"base_preamble": "Test snapshot preamble"},
            "snapshot": {"char_limit": 5000},
        }

    def test_snapshot_preserves_included_content(self):
        content = "def hello():\n    return 'world'\n"
        (self.root / "main.py").write_text(content, encoding="utf-8")

        snapshot = build_snapshot(self.root, config=self.config)

        self.assertIn(content, snapshot["snapshot"])
        record = next(
            item for item in snapshot["manifest"]["files"] if item["path"] == "main.py"
        )
        self.assertEqual(record["sha256"], hashlib.sha256(content.encode()).hexdigest())

    def test_excluded_file_is_not_included_as_content(self):
        (self.root / "private.py").write_text("SECRET_MARKER")
        snapshot = build_snapshot(self.root, config=self.config)

        self.assertNotIn("SECRET_MARKER", snapshot["snapshot"])
        record = next(
            item
            for item in snapshot["manifest"]["files"]
            if item["path"] == "private.py"
        )
        self.assertFalse(record["included"])

    def test_snapshot_is_written_as_markdown_and_json(self):
        (self.root / "main.py").write_text("pass\n")
        snapshot = build_snapshot(self.root, config=self.config)
        output_dir = self.root / ".izanagi"

        snapshot_path, manifest_path = write_snapshot(snapshot, output_dir)

        self.assertTrue(snapshot_path.is_file())
        self.assertTrue(manifest_path.is_file())
        self.assertIn("## Directory Structure", snapshot_path.read_text())
        manifest = json.loads(manifest_path.read_text())
        self.assertEqual(manifest["snapshot_id"], snapshot["manifest"]["snapshot_id"])


if __name__ == "__main__":
    unittest.main()
